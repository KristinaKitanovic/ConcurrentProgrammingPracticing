----------varijacija na readers writers problem ----------

U sistemu postoji jedna zajednička baza dokumenata. Bazi pristupaju tri tipa procesa:

čitaoci — mogu samo da čitaju dokumente
pisci — mogu da menjaju dokumente
recenzenti — pregledaju dokumente; oni ne menjaju sadržaj, ali njihov rad zahteva da u tom trenutku nema pisaca

Pravila su sledeća:

Više čitalaca može istovremeno pristupati bazi.
Više recenzenata može istovremeno pristupati bazi.
Čitaoci i recenzenti mogu biti unutra u isto vreme.
Pisac mora imati ekskluzivan pristup:
dok je pisac unutra, ne sme biti nijedan čitalac ni recenzent
dok postoje čitaoci ili recenzenti unutra, pisac mora čekati
Ako se pojavi pisac, novi čitaoci i novi recenzenti ne smeju ga beskonačno odlagati.

Napisati program korišćenjem semafora koji realizuje sinhronizaciju procesa:

Reader
Writer
Reviewer

Imamo sistem sa:

više proizvođača (producers) i više potrošača (consumers)
bafer ograničene veličine N
proizvođači proizvode dve vrste proizvoda:
tip A
tip B
Pravila:
Bafer je zajednički za oba tipa proizvoda (isti niz).
Consumeri imaju ograničenje:
ne smeju uzeti više od K uzastopnih proizvoda istog tipa
ako su uzeli K proizvoda tipa A, sledeći mora biti B (ako postoji), i obrnuto
Ako u baferu postoji samo jedan tip proizvoda, consumer:
mora da čeka dok se ne pojavi drugi tip (ako je dostigao limit K)
Proizvođači:
nasumično proizvode A ili B
ne smeju da prepišu bafer (klasično ograničenje)
Mora se obezbediti:
nema race condition-a
nema deadlock-a
nema starvation-a (nijedan tip ne sme biti zauvek ignorisan)

int waiting = 0;      // broj A/B consumera koji cekaju
sem cDone = 0;        // A/B cekaju ovde dok C ne zavrsi

typedef struct {
    int val;
    char type;
} element;

void producer(int id) {
    while (true) {
        element el = make_element();
        int c = rand() % 3;
        if (c == 0) el.type = 'A';
        else if (c == 1) el.type = 'B';
        else el.type = 'C';

        wait(empty);
        wait(mutexP);
            // ako je C, povecaj cCount pre ubacivanja
            if (el.type == 'C') {
                wait(mutexShared);
                    cCount++;
                signal(mutexShared);
            }
            buffer[tail] = el;
            tail = (tail + 1) % B;
            for (int i = 0; i < N; i++) signal(full[i]);
        signal(mutexP);
    }
}

void consumer(int id, char type) {
    int ri = 0;
    while (true) {
        wait(full[id]);

        // A i B provjeravaju da li ima C elemenata
        if (type == 'A' || type == 'B') {
            wait(mutexShared);
            while (cCount > 0) {//da nije while nego if moglo bi da se desi da oslobodimo nit od cDOne ali da ona ne krene da se izvrsava neko vreme i onda pocne kada opet ima C unutra zato ide while
                waiting++;
                signal(mutexShared);
                wait(cDone);        // cekaj da C consumeri zavrse
                wait(mutexShared);
                waiting--;
                
            }
                signal(mutexShared);
            
        }

        wait(mutexC[ri]);
            element el = null;
            if (buffer[ri].type == type) {
                el = buffer[ri];
            }

            cnt[ri]++;

            // ako je C consumer, smanji cCount
            if (type == 'C') {
                wait(mutexShared);
                    cCount--;
                    // ako su svi C consumeri uzeli, probudi A i B
                    if (cCount == 0) {
                        for (int i = 0; i < waiting; i++)
                            signal(cDone);
                    }
                signal(mutexShared);
            }

            if (cnt[ri] == N) {
                cnt[ri] = 0;
                signal(empty);
            }
        signal(mutexC[ri]);

        ri = (ri + 1) % B;
        if (el) consume(el);
    }
}

This solution does not generalize to the case where there is more than one car.
In order to do that, we have to satisfy some additional constraints:
• Only one car can be boarding at a time.
• Multiple cars can be on the track concurrently.
• Since cars can’t pass each other, they have to unload in the same order
they boarded.
• All the threads from one carload must disembark before any of the threads
from subsequent carloads.
Puzzle: modify the previous solution to handle the additional constraints.
You can assume that there are m cars, and that each car has a local variable
named i that contains an identifier between 0 and m − 1.

--------------------room party problem -----------------
1. Any number of students can be in a room at the same time.
2. The Dean of Students can only enter a room if there are no students in
the room (to conduct a search) or if there are more than 50 students in
the room (to break up the party).
3. While the Dean of Students is in the room, no additional students may
enter, but students may leave.
4. The Dean of Students may not leave the room until all students have left.
5. There is only one Dean of Students, so you do not have to enforce exclusion
among multiple deans.

void bus(){
    wait(mutex); 
    if(waiting > 0)
    {
        canBoard = min(50 - cnt, waiting);
        for(int i = 0; i < canBoard; i++)
        {
            signal(waitingRiders.pop())
            waiting--; 
        } 
        signal(mutex); 

        barrier();   
     
    }
    else signal(mutex); 
    depart(); 
}


void barrier(){
    wait(door1); 
    inbarrier++; 
    if(inbarrier == canBoard + 1)
    {
        signal(door2); 
    }
    else signal(door1); 
    wait(door2); 
    inbarrier--; 
    if(inbarrier ==0 )
    {
        signal(door1); 
    }
    else{
        signal(door2); 
    }
}

int students = 0; 
sem canEnter = 0; 
sem canLeave = 0; 
sem mutex = 1; 
int readyToLeave = 0; 
int readyToEnter = 0; 


void student(){
    getFood(); 
    wait(mutex); 
    readyToEnter++; 
    if(readyToLeave == 1)
    {
        //neko hoce da izadje
        readyToLeave--; 
        students--; 
        signal(canLeave); 
        readyToEnter--; 
        students++; 
        signal(mutex); 
    }
    if(readyToEnter == 1 && students == 0)
    {
        signal(mutex); 
        wait(canEnter); 
    }
    else if(readyToEnter == 2 and students == 0){
        //ulazim ja i dozvoljavam blokiranm da udje 
        students+=2; 
        readyToEnter-=2; 
        signal(canEnter); 

    }
    else {
        //vec ima studenata unutra 
        readyToEnter--; 
        students++; 
        signal(mutex); 
    }


    dine(); 


    wait(mutex); 
    readyToLeave++; 
    if(students == 2 && readyToLeave == 1)
    {
        //moramo da cekamo da zavrsi i onaj drugi 
        signal(mutex); 
        wait(canLeave)

    }
    else if(students == 2 and readyToLeave == 2)
    {
        //mozemo da izadjemo obojica 
        students-=2; 
        readyToLeave-=2; 
        signal(canLeave)
        signal(mutex); 
    }
    else{
        students--; 
        readyToLeave--; 
        signal(mutex); 
    }
    
}
